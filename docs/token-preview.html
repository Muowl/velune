/// <summary>Loads a theme without changing editor preferences.</summary>
public sealed class ThemeService(IThemeStore<Theme> store)
{
    private const int RetryLimit = 3;
    public event Action<Theme>? Changed;
    public string Name { get; private set; } = "Velune";

    [Obsolete("Use LoadAsync instead.")]
    public string Describe() => $"Theme: {Name}\n";

    public async Task<Theme?> LoadAsync(CancellationToken cancellationToken = default)
    {
        var path = @"C:\Themes\Velune";
        var json = """{"theme":"Velune","dark":true}""";
        for (var attempt = 0; attempt < RetryLimit; attempt++)
        {
            var theme = await store.FindAsync(Name, cancellationToken);
            if (theme is { Mode: ThemeMode.Dark })
            {
                Name = theme.Name;
                Changed?.Invoke(theme);
                Console.WriteLine($"{path}: {json}; {theme.Name}");
                return theme;
            }
        }
        return null;
    }
}

enum class ThemeMode { Dark, System };
struct Accent { unsigned char red, green, blue; };

template <typename T>
concept Named = requires(const T& value) {
    { value.name() } -> std::convertible_to<std::string_view>;
};

class Theme final {
public:
    explicit Theme(std::string name) : name_(name) {}
    [[nodiscard]] std::string_view name() const noexcept { return name_; }
    static constexpr int version = 1;
    ThemeMode mode = ThemeMode::Dark;
private:
    std::string name_;
};

template <Named T>
[[nodiscard]] std::string describe(const T& theme) {
    return std::string{theme.name()} + "\n";
}

using ThemePtr = std::unique_ptr<Theme>;
constexpr Accent accent{225, 76, 111};
constexpr auto metadata = R"json({"theme":"Velune","dark":true})json";

int preview() {
    ThemePtr theme = std::make_unique<Theme>("Velune");
    const auto label = describe(*theme);
    const auto version = VELUNE_VERSION;
    std::vector<int> levels{1, 2, 3};
    auto brighten = [](int value) -> int { return value + 1; };
    for (const auto& level : levels) {
        if (brighten(level) > 2 && theme->mode == ThemeMode::Dark) {
            return static_cast<int>(label.size() + version[0]);
        }
    }
    return 0;
}

use std::fmt;

// Comments should remain readable and upright.
const THEME_NAME: &str = "Velune";
static ENABLED: bool = true;

#[derive(Debug, Clone, Copy, PartialEq)]
pub enum ThemeMode {
    Dark,
    System,
}

/// A borrowed theme with a named lifetime.
#[derive(Debug)]
pub struct Theme<'a, const N: usize> {
    pub name: &'a str,
    pub accent: [u8; N],
    pub mode: ThemeMode,
}

pub trait Named {
    fn name(&self) -> &str;
}

impl<'a, const N: usize> Named for Theme<'a, N> {
    fn name(&self) -> &str {
        self.name
    }
}

impl<const N: usize> fmt::Display for Theme<'_, N> {
    fn fmt(&self, formatter: &mut fmt::Formatter<'_>) -> fmt::Result {
        write!(formatter, "Theme: {}\n", self.name)
    }
}

pub type ThemeResult<T> = Result<T, &'static str>;

macro_rules! theme_label {
    ($value:expr) => { format!("[{}]", $value) };
}

pub async fn load_theme<'a>(name: &'a str) -> ThemeResult<Theme<'a, 3>> {
    if name.is_empty() {
        return Err("Empty name");
    }
    Ok(Theme { name, accent: [225, 76, 111], mode: ThemeMode::Dark })
}

fn main() {
    let mut theme = Theme { name: THEME_NAME, accent: [225, 76, 111], mode: ThemeMode::Dark };
    let names = [THEME_NAME, "System"].map(|name| name.to_lowercase());
    let json = r#"{"theme":"Velune","dark":true}"#;
    let separator = '\n';
    let marker = 'v';
    if ENABLED && names.iter().any(|name| !name.is_empty()) {
        theme.mode = ThemeMode::System;
    }
    match theme.mode {
        ThemeMode::Dark => println!("{}", theme_label!(theme.name())),
        ThemeMode::System => println!("{json}{separator}{marker}"),
    }
}

package velune.preview;

import java.util.List;
import java.util.Optional;
import java.util.function.Function;

// Comments should remain readable and upright.
public final class ThemePreview {
    private static final String THEME_NAME = "Velune";
    private static final boolean ENABLED = true;

    public enum ThemeMode { DARK, SYSTEM }
    public record Accent(int red, int green, int blue) {}
    public record Theme(String name, Accent accent, ThemeMode mode) {}

    public interface ThemeStore<T> {
        Optional<T> find(String name);
    }

    /**
     * Returns the theme label.
     * @param theme the selected theme
     * @return a readable name
     * @see Theme
     */
    @Deprecated(since = "0.1", forRemoval = false)
    public static String describe(Theme theme) {
        return "Theme: " + theme.name() + "\n";
    }

    @SafeVarargs
    private static <T> List<T> entries(T... values) {
        return List.of(values);
    }

    public static void main(String[] args) {
        var theme = new Theme(THEME_NAME, new Accent(225, 76, 111), ThemeMode.DARK);
        Function<Theme, String> label = value -> value.name().toLowerCase();
        var names = entries(theme).stream().map(Theme::name).toList();
        Optional<? extends Theme> selected = Optional.of(theme);
        var json = """
                {"theme":"Velune","dark":true}
                """;
        char marker = 'v';
        var mode = switch (theme.mode()) {
            case DARK -> "dark";
            case SYSTEM -> "system";
        };
        if (ENABLED && !names.isEmpty() && selected.isPresent()) {
            System.out.println(label.apply(theme) + json + marker + mode);
        }
    }
}
