// Tokens e estruturas comuns em TypeScript.
interface Theme {
  name: string;
  dark: boolean;
  background: string;
}

export const theme: Theme = {
  name: "Velune",
  dark: true,
  background: "#141119",
};

export function alpha(value: number): number {
  return Math.min(1, Math.max(0, value));
}

type Palette<T> = Readonly<Record<string, T>>;

// Funções, operadores, templates e expressões regulares.
export async function loadTheme(name = "Velune") {
  const theme = { name, dark: true, opacity: 0.8 };
  const isValid = /^#[0-9a-f]{6}$/i.test("#141119");
  return { ...theme, isValid, label: `Tema ${name}` };
}

loadTheme().then(theme => console.log(theme.name));

import { useState } from "react";

  const [active, setActive] = useState(false);
  return (
    <button
      className="theme-card"
      aria-pressed={active}
      onClick={() => { setActive(true); onSelect(name); }}
    >
      <span>{name}</span>
      {active ? "Selecionado" : "Selecionar"}
    </button>
  );
}

from dataclasses import dataclass

# Comentários sem itálico automático.
@dataclass(frozen=True)
class Theme:
    name: str
    dark: bool
    background: str


def alpha(value: float) -> float:
    """Limita a opacidade ao intervalo entre zero e um."""
    return min(1, max(0, value))


theme = Theme(name="Velune", dark=True, background="#141119")
message = f"Tema {theme.name}: {alpha(0.8)}"
print(message)
